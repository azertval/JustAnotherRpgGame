43[[nodiscard]] inline std::vector<GridPosition> footprintCells(GridPosition anchor,
44 PieceFootprint footprint) {
45 const int columns = footprint.columns < 1 ? 1 : footprint.columns;
46 const int rows = footprint.rows < 1 ? 1 : footprint.rows;
47 std::vector<GridPosition> cells;
48 cells.reserve(static_cast<std::size_t>(columns) * static_cast<std::size_t>(rows));
49 for (int row = 0; row < rows; ++row) {
50 for (int column = 0; column < columns; ++column) {
51 cells.push_back(
52 GridPosition{.column = anchor.column + column, .row = anchor.row + row});
53 }
54 }
55 return cells;
