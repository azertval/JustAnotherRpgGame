90[[nodiscard]] inline SpriteQuad standingPieceQuad(const SceneTexture& texture,
91 core::Vector2 topVertex,
92 float unitsPerScenePixel) {
93 SpriteQuad quad;
94 const auto height = static_cast<float>(texture.height);
95 const core::Vector2 anchor = texture.anchor.value_or(
96 core::Vector2{static_cast<float>(SCENE_HALF_TILE_WIDTH_PIXELS),
97 height - static_cast<float>(SCENE_TILE_HEIGHT_PIXELS)});
98 quad.x = topVertex.x - anchor.x * unitsPerScenePixel;
99 quad.y = topVertex.y - anchor.y * unitsPerScenePixel;
100 quad.width = static_cast<float>(texture.width) * unitsPerScenePixel;
101 quad.height = height * unitsPerScenePixel;
102 return quad;
103}
