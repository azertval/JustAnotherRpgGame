90 explicit IsoView(const core::IsoProjection& projection) noexcept
91 : _projection(projection),
92 _sine(std::clamp(projection.tileHeight() / projection.tileWidth(), 0.001F, MAXIMUM_SINE)),
93 _cosine(std::sqrt(1.0F - (_sine * _sine))),
94 _groundY(projection.origin().y) {}
95
97 [[nodiscard]] float sine() const noexcept {
98 return _sine;

101 [[nodiscard]] float cosine() const noexcept {
102 return _cosine;

107 [[nodiscard]] float unitsPerMetre() const noexcept {
108 return _projection.tileWidth() / (core::METERS_PER_TILE * std::numbers::sqrt2_v<float>);
109 }
110
112 [[nodiscard]] float groundDepth(float y) const noexcept {
113 return -(y - _groundY) * _cosine / _sine;
114 }
115
118 [[nodiscard]] float raisedDepth(float groundY, float rise) const noexcept {
119 return groundDepth(groundY) - (rise * _sine / _cosine);
120 }
121
126 [[nodiscard]] float standingDepth(float footY, float y) const noexcept {
127 return y < footY ? raisedDepth(footY, footY - y) : groundDepth(y);
128 }
129
131 [[nodiscard]] float imageBias() const noexcept {
132 return IMAGE_DEPTH_BIAS_TILES * _projection.tileWidth();
133 }

136 [[nodiscard]] float riseOf(float metres) const noexcept {
137 return metres * unitsPerMetre() * _cosine;

144 [[nodiscard]] ViewTransform meshTransform(core::Vector2 gridPoint, float rise) const noexcept {
145 const core::Vector2 origin = _projection.gridToWorld(gridPoint);
146 // Un mètre le long d'une colonne ou d'une ligne : un tiers de demi-losange par 0,5 m.
147 const float along = _projection.tileWidth() / (2.0F * core::METERS_PER_TILE);
148 const float up = unitsPerMetre();
149 return ViewTransform{along,
150 0.0F,
151 -along,
152 origin.x,
153 along * _sine,
154 -up * _cosine,
155 along * _sine,
156 origin.y - rise,
157 -along * _cosine,
158 -up * _sine,
159 -along * _cosine,
160 raisedDepth(origin.y, rise)};
161 }
162
169 [[nodiscard]] static ViewTransform turned(const ViewTransform& transform, float yaw) noexcept {
170 const float cosine = std::cos(yaw);
171 const float sine = std::sin(yaw);
172 ViewTransform out = transform;
173 for (std::size_t row = 0; row < 3; ++row) {
174 const float alongX = transform[row * 4];
175 const float alongZ = transform[(row * 4) + 2];
176 out[row * 4] = (alongX * cosine) - (alongZ * sine);
177 out[(row * 4) + 2] = (alongX * sine) + (alongZ * cosine);
178 }
179 return out;
180 }
181
183 [[nodiscard]] static std::array<float, 3> apply(const ViewTransform& transform, float x,
184 float y, float z) noexcept {
185 return {(transform[0] * x) + (transform[1] * y) + (transform[2] * z) + transform[3],
186 (transform[4] * x) + (transform[5] * y) + (transform[6] * z) + transform[7],
187 (transform[8] * x) + (transform[9] * y) + (transform[10] * z) + transform[11]};
188 }
189
194 [[nodiscard]] static core::Rect projectedBounds(const ViewTransform& transform,
195 const std::array<float, 3>& minimum,
196 const std::array<float, 3>& maximum) noexcept {
197 float left = 0.0F;
198 float top = 0.0F;
199 float right = 0.0F;
200 float bottom = 0.0F;
201 for (int corner = 0; corner < 8; ++corner) {
202 const std::array<float, 3> point =
203 apply(transform, (corner & 1) != 0 ? maximum[0] : minimum[0],
204 (corner & 2) != 0 ? maximum[1] : minimum[1],
205 (corner & 4) != 0 ? maximum[2] : minimum[2]);
206 if (corner == 0) {
207 left = right = point[0];
208 top = bottom = point[1];
209 continue;
210 }
211 left = std::min(left, point[0]);
212 right = std::max(right, point[0]);
213 top = std::min(top, point[1]);
214 bottom = std::max(bottom, point[1]);
215 }
216 return core::Rect{{left, top}, {right - left, bottom - top}};
217 }
218
220 [[nodiscard]] DepthRange depthRange() const noexcept {
221 const float margin = DEPTH_MARGIN_TILES * _projection.tileWidth();
222 return DepthRange{.nearest = raisedDepth(_projection.sceneSize().y + margin, margin),
223 .farthest = groundDepth(-margin)};
224 }
225
226private:
227 core::IsoProjection _projection;
228 float _sine;
229 float _cosine;
231 float _groundY;
232};
