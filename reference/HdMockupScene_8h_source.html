43inline hmi::WorldSceneSnapshot hdMockupSnapshot(const nlohmann::json& scene,
44 const std::filesystem::path& directory) {
45 hmi::WorldSceneSnapshot snapshot;
46 snapshot.place = scene["place"].get<std::string>();
47 snapshot.diamondRatio = scene["diamondRatio"].get<float>();
48 snapshot.columns = scene["columns"].get<int>();
49 snapshot.rows = scene["rows"].get<int>();
50 const auto cells =
51 static_cast<std::size_t>(snapshot.columns) * static_cast<std::size_t>(snapshot.rows);
52 snapshot.floors.assign(cells, std::string{});
53 snapshot.relief.assign(cells, std::string{});
54 snapshot.types.assign(cells, core::TileType::Solid);
55 snapshot.reliefTypes.assign(cells, core::TileType::Empty);
56
57 const nlohmann::json& legend = scene["legend"];
58 const nlohmann::json& floors = scene["floors"];
59 for (int row = 0; row < snapshot.rows; ++row) {
60 const std::string line = floors[static_cast<std::size_t>(row)].get<std::string>();
61 for (int column = 0; column < snapshot.columns; ++column) {
62 const std::string symbol(1, line[static_cast<std::size_t>(column)]);
63 snapshot.floors[(static_cast<std::size_t>(row) * snapshot.columns) + column] =
64 legend[symbol].get<std::string>();
65 }
66 }
67
68 const core::ScenePieceManifestResult manifest = core::ScenePieceManifest::loadFromFile(
69 directory / "Scene" / snapshot.place / "manifest.json");
70 // Un manifeste illisible laisse les pièces sans emprise : la comparaison d'images le verra.
71 for (const nlohmann::json& piece : scene["pieces"]) {
72 const std::string name = piece["piece"].get<std::string>();
73 const int column = piece["column"].get<int>();
74 const int row = piece["row"].get<int>();
75 snapshot.relief[(static_cast<std::size_t>(row) * snapshot.columns) + column] = name;
76 if (const core::ScenePiece* const declared =
77 manifest.ok() ? manifest.manifest.find(name) : nullptr) {
78 snapshot.footprints.insert_or_assign(name, declared->footprint());
79 }
80 }
81 return snapshot;
82}
