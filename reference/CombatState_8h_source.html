129[[nodiscard]] constexpr bool crossedBelow(const CombatEvent& event, int numerator,
130 int denominator) noexcept {
131 const long long seuil = static_cast<long long>(event.maximumHitPoints) * numerator;
132 return static_cast<long long>(event.hitPointsBefore) * denominator >= seuil &&
133 static_cast<long long>(event.hitPointsAfter) * denominator < seuil;
134}
