124[[nodiscard]] constexpr bool crossedBelow(const CombatEvent& event, int numerator,
125 int denominator) noexcept {
126 const long long seuil = static_cast<long long>(event.maximumHitPoints) * numerator;
127 return static_cast<long long>(event.hitPointsBefore) * denominator >= seuil &&
128 static_cast<long long>(event.hitPointsAfter) * denominator < seuil;
129}
