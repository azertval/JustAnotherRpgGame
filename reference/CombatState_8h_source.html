125[[nodiscard]] constexpr bool crossedBelow(const CombatEvent& event, int numerator,
126 int denominator) noexcept {
127 const long long seuil = static_cast<long long>(event.maximumHitPoints) * numerator;
128 return static_cast<long long>(event.hitPointsBefore) * denominator >= seuil &&
129 static_cast<long long>(event.hitPointsAfter) * denominator < seuil;
130}
