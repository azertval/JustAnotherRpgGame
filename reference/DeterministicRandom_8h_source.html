106 [[nodiscard]] int nextInt(int min, int max) noexcept {
107 if (max <= min) {
108 return min;
109 }
110 const std::uint32_t etendue = static_cast<std::uint32_t>(max - min) + 1U;
111 // `2^32 mod etendue` : le nombre de valeurs en trop, celles qui n'ont pas de contrepartie
112 // sur tout l'intervalle. On rejette la queue BASSE, `[0, reste[`, et non la haute.
113 //
114 // Rejeter la queue haute serait plus naturel à lire — « au-delà du dernier multiple
115 // complet » — mais ce seuil vaut `2^32` quand `etendue` divise `2^32`, et `2^32` ne tient
116 // pas dans un `std::uint32_t` : il retombe à 0, la condition devient toujours vraie, et la
117 // boucle ne se termine jamais. Le défaut ne se voit que sur les **puissances de deux** :
118 // un d6 et un d20 passent, un d8 bloque. C'est un test de rejouabilité sur 3d8 qui l'a
119 // trouvé, pas une relecture.
120 const std::uint32_t reste = (std::uint32_t{0} - etendue) % etendue;
121 std::uint32_t tirage = nextUInt32();
122 while (tirage < reste) {
123 tirage = nextUInt32();
124 }
125 return min + static_cast<int>(tirage % etendue);
126 }
127
128private:
129 std::uint64_t _state;
130};
