65[[nodiscard]] inline float artTileWidth(const SceneTexture& texture) noexcept {
66 if (texture.artTile.x > 0.0F) {
67 return texture.artTile.x;
68 }
69 if (texture.frameHeight > 0) {
70 return static_cast<float>(texture.frameHeight);
71 }
72 return static_cast<float>(

137[[nodiscard]] inline SpriteQuad standingPieceQuad(const SceneTexture& texture,
138 core::Vector2 topVertex, float tileWidth,
139 float ratio) {
140 SpriteQuad quad;
141 const float art = artTileWidth(texture);
142 const float unitsPerPixel = tileWidth / art;
143 const auto height = static_cast<float>(texture.height);
144 const core::Vector2 anchor =
145 texture.anchor.value_or(core::Vector2{art / 2.0F, height - artTileHeight(texture, ratio)});
146 quad.x = topVertex.x - anchor.x * unitsPerPixel;
147 quad.y = topVertex.y - anchor.y * unitsPerPixel;
148 quad.width = static_cast<float>(texture.width) * unitsPerPixel;
149 quad.height = height * unitsPerPixel;
150 return quad;
151}

161inline constexpr float FLOOR_SEAM_OVERLAP = 1.0F / 256.0F;

167[[nodiscard]] inline SpriteQuad floorQuad(const core::Rect& bounds) noexcept {
168 // Le meme debord relatif sur les deux axes : le losange grandit sans changer de rapport.
169 const float grow = FLOOR_SEAM_OVERLAP * 2.0F;
170 SpriteQuad quad;
171 quad.width = bounds.size.x * (1.0F + grow);
172 quad.height = bounds.size.y * (1.0F + grow);
173 quad.x = bounds.position.x - ((quad.width - bounds.size.x) / 2.0F);
174 quad.y = bounds.position.y - ((quad.height - bounds.size.y) / 2.0F);
175 return quad;
176}

197[[nodiscard]] inline SpriteQuad figureQuad(const SceneTexture& texture, int frame, float centerX,
198 float bottomY, float tileWidth) {
199 const float unitsPerPixel = tileWidth / artTileWidth(texture);
200 const int frameWidth = frameWidthOf(texture);
201 const int frameHeight = frameHeightOf(texture);
202 SpriteQuad quad;
203 quad.width = static_cast<float>(frameWidth) * unitsPerPixel;
204 quad.height = static_cast<float>(frameHeight) * unitsPerPixel;
205 quad.x = centerX - (quad.width / 2.0F);
206 quad.y = bottomY - quad.height;
207 if (texture.width > 0 && texture.height > 0) {
208 const auto width = static_cast<float>(texture.width);
209 quad.u0 = static_cast<float>(frame * frameWidth) / width;
210 quad.u1 = static_cast<float>((frame + 1) * frameWidth) / width;
211 quad.v0 = 0.0F;
212 quad.v1 = static_cast<float>(frameHeight) / static_cast<float>(texture.height);
213 }
214 return quad;
215}
