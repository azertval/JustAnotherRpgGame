136 template <typename T>
137 ComponentPool<T>& poolFor() {
138 const std::type_index key(typeid(T));
139 auto iterator = _pools.find(key);
140 if (iterator == _pools.end()) {
141 iterator = _pools.emplace(key, std::make_unique<ComponentPool<T>>()).first;
142 }
143 return *static_cast<ComponentPool<T>*>(iterator->second.get());
144 }
145
151 template <typename T>
152 const ComponentPool<T>* findPool() const {
153 const auto iterator = _pools.find(std::type_index(typeid(T)));
154 if (iterator == _pools.end()) {
155 return nullptr;
156 }
157 return static_cast<const ComponentPool<T>*>(iterator->second.get());
158 }
