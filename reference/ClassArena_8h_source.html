52inline const RpgCatalogs& rpgCatalogs() {
53 static const RpgCatalogs catalogs = [] {
54 const std::filesystem::path rpg{JADG_RPG_DIR};
55 RpgCatalogs read;
56 read.options = core::loadCharacterOptions(rpg);
57 read.experience = core::loadExperienceTable(rpg / "rules" / "experience.json");
58 read.rules = core::loadCharacterCreationRules(rpg / "rules" / "character-creation.json");
59 read.equipment = core::loadEquipment(rpg / "weapons", rpg / "armors");
60 read.items = core::loadItems(rpg / "items");
61 read.encumbrance = core::loadEncumbranceRules(rpg / "rules" / "encumbrance.json");
62 return read;
63 }();
64 return catalogs;
65}

79inline std::vector<std::string> levelUpTo(core::CharacterSheet& sheet, int level) {
80 const RpgCatalogs& catalogs = rpgCatalogs();
81 std::vector<std::string> missing;
82 const core::PlayableClass* playableClass = catalogs.options.findClass(sheet.classId);
83 if (playableClass == nullptr) {
84 missing.push_back(sheet.classId);
85 return missing;
86 }
87 const int threshold = catalogs.experience.thresholdAt(level);
88 if (threshold > sheet.experiencePoints) {
89 core::gainExperience(sheet, catalogs.experience, playableClass->hitDie,
90 threshold - sheet.experiencePoints);
91 }
92 core::applyClassFeatures(sheet, *playableClass, catalogs.options, catalogs.rules, missing);
93 return missing;
94}

120inline core::AttackProfile weaponAttack(const core::CharacterSheet& sheet, const char* weaponId) {
121 const RpgCatalogs& catalogs = rpgCatalogs();
122 const core::Weapon* weapon = catalogs.equipment.findWeapon(weaponId);
123 if (weapon == nullptr) {
124 return core::weaponAttackFor(sheet, nullptr,
125 core::proficiencyBonus(sheet, catalogs.experience));
126 }
127 return core::weaponAttackFor(sheet, weapon, core::proficiencyBonus(sheet, catalogs.experience),
128 core::isProficientWith(sheet, *weapon));
129}

137inline core::ArenaContestant hero(const core::CharacterSheet& sheet,
138 const core::Inventory& inventory,
139 std::vector<core::AttackProfile> attacks,
140 core::GridPosition place,
141 core::CombatSide side = core::CombatSide::Allies,
142 int initiative = 100) {
143 const RpgCatalogs& catalogs = rpgCatalogs();
144 core::CombatantProfile profile =
145 core::profileFor(sheet, side, core::proficiencyBonus(sheet, catalogs.experience));
146 profile.armorClass = armorClassOf(sheet, inventory);
147 profile.initiativeModifier = initiative;
148 std::vector<core::ArenaSpell> spells;
149 if (const core::PlayableClass* playableClass = catalogs.options.findClass(sheet.classId)) {
150 std::vector<std::string> skipped;
151 spells = core::arenaSpellsFor(sheet, *playableClass, catalogs.options.spells,
152 core::proficiencyBonus(sheet, catalogs.experience), skipped);
153 }
154 return {.profile = std::move(profile),
155 .attacks = std::move(attacks),
156 .position = place,
157 .markId = {},
158 .behavior = {},
159 .capacities = sheet.capacities,
160 .spells = std::move(spells)};
161}
