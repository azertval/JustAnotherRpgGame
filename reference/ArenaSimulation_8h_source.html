94 [[nodiscard]] hmi::HeroContestantSource sourceOf(
95 const core::LoadedCharacterSheet& member) const {
96 const core::ItemLookup lookup{.items = &items, .equipment = &equipment};
97 hmi::HeroContestantSource hero{
98 .sheet = member.sheet,
99 .proficiency = core::proficiencyBonus(member.sheet, experience),
100 .armorClass =
101 core::derivedStatsFor(member.sheet, member.inventory, lookup, rules, encumbrance)
102 .armorClass,
103 .weapon = std::nullopt,
104 .spells = {}};
105 if (const core::Weapon* weapon =
106 equipment.findWeapon(member.inventory.at(core::EquipmentSlot::MainHand))) {
107 hero.weapon = *weapon;
108 }
109 if (const core::PlayableClass* playableClass = options.findClass(member.sheet.classId)) {
110 std::vector<std::string> skipped;
111 hero.spells = core::arenaSpellsFor(member.sheet, *playableClass, options.spells,
112 hero.proficiency, skipped);
113 }
114 return hero;
115 }

131[[nodiscard]] inline Heros loadHeroes(const std::filesystem::path& elements,
132 const std::vector<std::string>& members, int level = 1) {
133 const std::filesystem::path rpg = elements / "Rpg";
134 Heros heros;
135 heros.options = core::loadCharacterOptions(rpg);
136 heros.skills = core::loadSkills(rpg / "skills");
137 heros.experience = core::loadExperienceTable(rpg / "rules" / "experience.json");
138 heros.rules = core::loadCharacterCreationRules(rpg / "rules" / "character-creation.json");
139 for (const std::string& membre : members) {
140 core::LoadedCharacterSheet charge = core::loadCharacterSheet(
141 rpg / "characters" / (membre + ".json"), heros.options, heros.rules, heros.experience);
142 if (level > charge.sheet.level) {
143 if (const core::PlayableClass* classe = heros.options.findClass(charge.sheet.classId)) {
144 std::vector<std::string> manquants;
145 static_cast<void>(core::levelUpTo(charge.sheet, level, *classe, heros.options,
146 heros.rules, heros.experience, manquants));
147 }
148 }
149 heros.members.push_back(std::move(charge));
150 }
151 heros.items = core::loadItems(rpg / "items");
152 heros.equipment = core::loadEquipment(rpg / "weapons", rpg / "armors");
153 heros.encumbrance = core::loadEncumbranceRules(rpg / "rules" / "encumbrance.json");
154 return heros;
155}

173[[nodiscard]] inline core::GridPosition triggerOf(const core::Level& map,
174 std::string_view encounterId) {
175 for (const core::MapEntity& entity : map.entities()) {
176 if (entity.type != "encounter") {
177 continue;
178 }
179 const auto id = entity.properties.find("encounterId");
180 if (id != entity.properties.end()) {
181 if (const auto* text = std::get_if<std::string>(&id->second);
182 text != nullptr && *text == encounterId) {
183 return entity.position;
184 }
185 }
186 }
187 return MAITRE;
188}

194[[nodiscard]] inline std::optional<core::CombatOutcome> playEncounter(
195 const core::Level& map, const ArenaContent& arena, std::string_view encounterId,
196 const std::vector<hmi::HeroContestantSource>& party, std::uint64_t seed,
197 core::EncounterRun* engaged = nullptr) {
198 const core::Encounter* const rencontre = arena.encounters.find(encounterId);
199 if (rencontre == nullptr) {
200 ADD_FAILURE() << "rencontre inconnue : " << encounterId;
201 return std::nullopt;
202 }
203 // Le meneur devant le maitre, ses suiveurs dans son dos, vers l'est.
204 std::vector<core::GridPosition> cases;
205 for (std::size_t rang = 0; rang < party.size(); ++rang) {
206 cases.push_back({.column = DEVANT_LE_MAITRE.column + static_cast<int>(rang),
207 .row = DEVANT_LE_MAITRE.row});
208 }
209 const core::MapEncounterResult prepared =
210 core::prepareMapEncounter(map, SABLE, *rencontre, triggerOf(map, encounterId), cases,
211 core::ExplorationSnapshot{}, "");
212 if (!prepared.ok()) {
213 ADD_FAILURE() << prepared.issue;
214 return std::nullopt;
215 }
216 const core::MapEncounterSetup& setup = *prepared.setup;
217 if (engaged != nullptr) {
218 *engaged = setup.run;
219 }
220 core::ArenaSession session(setup.battlefield);
221 session.setOpportunityPolicy(core::aiOpportunityPolicy(arena.behaviors));
222 core::ArenaBout bout{.contestants = {},
223 .seed = seed,
224 .lethal = true,
225 .heroicMark = false,
226 .flanking = true,
227 .escapable = setup.run.escapable};
228 for (std::size_t rang = 0; rang < party.size() && rang < setup.partyCells.size(); ++rang) {
229 core::ArenaContestant membre = hmi::heroContestant(party[rang], core::CombatSide::Allies);
230 membre.position = setup.partyCells[rang];
231 // Le joueur ne joue pas : l'IA tient la place de chacun, sorts et soins compris.
232 membre.behavior = profileOfClass(party[rang].sheet.classId);
233 bout.contestants.push_back(std::move(membre));
234 }
235 for (const core::CombatantPlacement& placement : setup.run.placements) {
236 const core::Creature* const creature = arena.bestiary.find(placement.creatureId);
237 if (creature == nullptr) {
238 ADD_FAILURE() << "creature inconnue : " << placement.creatureId;
239 return std::nullopt;
240 }
241 core::ArenaContestant enemy =
242 hmi::creatureContestant(*creature, core::CombatSide::Enemies, &arena.behaviors);
243 enemy.position = placement.position;
244 bout.contestants.push_back(std::move(enemy));
245 }
246 const core::ArenaMount mount = session.mount(bout);
247 if (mount.allies.empty() || mount.enemies.empty() || !session.start()) {
248 ADD_FAILURE() << "montage refuse a la graine " << seed;
249 return std::nullopt;
250 }
251 for (int tours = 0; tours < 600; ++tours) {
252 if (session.combat().phase() == core::CombatPhase::Ended) {
253 return session.outcome();
254 }
255 if (!core::playTurn(session, arena.behaviors)) {
256 return std::nullopt;
257 }
258 }
259 return std::nullopt;
260}
