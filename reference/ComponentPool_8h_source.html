60template <typename T>
61class ComponentPool : public IComponentPool {
62public:
64 static constexpr std::size_t INVALID_POSITION = ~std::size_t{0};
65
72 void add(Entity entity, const T& component) {
73 JADG_ASSERT(!has(entity), "L'entite possede deja ce composant.");
74
75 ensureSparseSize(entity.index);
76 _sparse[entity.index] = _components.size();
77 _components.push_back(component);
78 _entities.push_back(entity);
79 }
80
86 void remove(Entity entity) {
87 JADG_ASSERT(has(entity), "L'entite ne possede pas ce composant.");
88
89 const std::size_t removed = _sparse[entity.index];
90 const std::size_t last = _components.size() - 1;
91
92 // Swap-and-pop : on remonte le dernier élément à la place du supprimé
93 // pour garder le tableau dense contigu, puis on met à jour le creux de
94 // l'entité déplacée pour qu'il pointe vers sa nouvelle position.
95 _components[removed] = _components[last];
96 _entities[removed] = _entities[last];
97 _sparse[_entities[removed].index] = removed;
98
99 _components.pop_back();
100 _entities.pop_back();
101 _sparse[entity.index] = INVALID_POSITION;
102 }
103
109 bool removeIfPresent(Entity entity) override {
110 if (!has(entity)) {
111 return false;
112 }
113 remove(entity);
114 return true;
115 }
116
122 [[nodiscard]] bool has(Entity entity) const {
123 if (entity.index >= _sparse.size()) {
124 return false;
125 }
126 const std::size_t position = _sparse[entity.index];
127 // Position valide et pointant une entrée dont l'entité correspond
128 // exactement (l'égalité inclut la génération : un handle périmé échoue).
129 return position != INVALID_POSITION && _entities[position] == entity;
130 }
131
138 [[nodiscard]] T& get(Entity entity) {
139 JADG_ASSERT(has(entity), "L'entite ne possede pas ce composant.");
140 return _components[_sparse[entity.index]];
141 }
142
149 [[nodiscard]] const T& get(Entity entity) const {
150 JADG_ASSERT(has(entity), "L'entite ne possede pas ce composant.");
151 return _components[_sparse[entity.index]];
152 }
153
155 [[nodiscard]] std::size_t size() const {
156 return _components.size();
157 }
158
160 [[nodiscard]] bool empty() const {
161 return _components.empty();
162 }
163
168 [[nodiscard]] const std::vector<Entity>& entities() const {
169 return _entities;
170 }
171
176 [[nodiscard]] const std::vector<T>& components() const {
177 return _components;
178 }
179
180private:
185 void ensureSparseSize(Entity::Index index) {
186 if (index >= _sparse.size()) {
187 _sparse.resize(static_cast<std::size_t>(index) + 1, INVALID_POSITION);
188 }
189 }
190
192 std::vector<T> _components;
194 std::vector<Entity> _entities;
196 std::vector<std::size_t> _sparse;
197};
